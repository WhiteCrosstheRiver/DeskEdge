cmake_minimum_required(VERSION 3.24)
project(DeskEdge VERSION 1.3.0 LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32)
  message(FATAL_ERROR "DeskEdge requires Windows 10 or later.")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
add_library(deskedge_core STATIC src/core.cpp src/grid.cpp)
target_include_directories(deskedge_core PUBLIC src vendor/nlohmann-json/include)
target_link_libraries(deskedge_core PUBLIC ole32 shell32 shlwapi)
target_compile_options(deskedge_core PRIVATE /W4 /utf-8 /permissive-)
add_library(imgui STATIC vendor/imgui/imgui.cpp vendor/imgui/imgui_draw.cpp vendor/imgui/imgui_tables.cpp vendor/imgui/imgui_widgets.cpp vendor/imgui/backends/imgui_impl_win32.cpp vendor/imgui/backends/imgui_impl_dx11.cpp vendor/imgui/misc/cpp/imgui_stdlib.cpp)
target_include_directories(imgui PUBLIC vendor/imgui vendor/imgui/backends vendor/imgui/misc/cpp)
target_compile_options(imgui PRIVATE /utf-8)
target_link_libraries(imgui PUBLIC d3d11 dxgi dwmapi imm32)
add_executable(DeskEdge WIN32 src/main.cpp src/platform.cpp src/icons.cpp src/material.cpp src/motion.cpp src/shell.cpp src/drop.cpp src/ui.cpp src/ui_tests.cpp src/app.rc)
target_link_libraries(DeskEdge PRIVATE deskedge_core imgui comctl32 shcore windowscodecs psapi dbghelp dcomp)
target_compile_options(DeskEdge PRIVATE /W4 /utf-8 /permissive- /Zi)
target_compile_options(imgui PRIVATE /Zi)
target_link_options(DeskEdge PRIVATE /DEBUG)
target_link_options(DeskEdge PRIVATE /MANIFEST:EMBED "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/src/app.manifest")
set_target_properties(DeskEdge PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
enable_testing()
add_executable(DeskEdgeTests tests/core_tests.cpp)
target_link_libraries(DeskEdgeTests PRIVATE deskedge_core)
target_compile_options(DeskEdgeTests PRIVATE /W4 /utf-8)
add_test(NAME DeskEdge.Core COMMAND DeskEdgeTests)
install(TARGETS DeskEdge RUNTIME DESTINATION .)
install(FILES README.md THIRD_PARTY_NOTICES.md DESTINATION .)
